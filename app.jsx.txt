ATHLETE: ${athlete.name || "Athlete"}, Age ${athlete.age || "?"}, ${athlete.bw ? athlete.bw + " lbs BW" : ""}, ${athlete.experience} level
GOALS: ${goals.join(", ")}

Return this exact JSON structure (fill in real numbers/text, no angle-bracket placeholders):

// ─── Render analysis ───────────────────────────────────────────────────────────
function AnalysisView({ data }) {
  const score = data.overallScore || 0;
  const scoreColor = score >= 75 ? "var(--success)" : score >= 50 ? "var(--accent)" : "var(--accent2)";

  const barColor = (status) =>
    status === "good" ? "var(--success)" : status === "warn" ? "var(--danger)" : "var(--accent)";

  return (
    <div className="analysis-wrap">
      {/* Header */}
      <div className="analysis-header">
        <div>
          <div className="analysis-title">COACHING REPORT</div>
          <div className="analysis-meta">
            {data.strengthAge} · Focus: {data.primaryFocus}
          </div>
        </div>
        <div className="rating-badge">
          <div className="rating-ring" style={{ border: `3px solid ${scoreColor}`, color: scoreColor }}>
            {score}
          </div>
          <div className="rating-label">Score</div>
        </div>
      </div>

      {/* Score cards */}
      <div className="score-grid">
        {["strength", "balance", "potential"].map(k => (
          <div className="score-card" key={k}>
            <div className="score-card-label">{k}</div>
            <div className="score-card-value" style={{
              color: data.scores[k] >= 75 ? "var(--success)" : data.scores[k] >= 50 ? "var(--accent)" : "var(--accent2)"
            }}>
              {data.scores[k]}
            </div>
            <div className="score-card-sub">{data.scoreNotes[k]}</div>
          </div>
        ))}
      </div>

      {/* Key findings */}
      <div className="analysis-section">
        <div className="analysis-section-header">
          <span className="section-icon">🔍</span> Key Findings
        </div>
        <div className="analysis-section-body">
          <div className="analysis-prose">{data.keyFindings}</div>
        </div>
      </div>

      {/* Ratios */}
      {data.ratios && data.ratios.some(r => r.value !== null) && (
        <div className="analysis-section">
          <div className="analysis-section-header">
            <span className="section-icon">⚖️</span> Lift Ratios
          </div>
          <div className="analysis-section-body">
            {data.ratios.filter(r => r.value !== null).map((r, i) => (
              <div className="ratio-bar-row" key={i}>
                <div className="ratio-bar-label">{r.label}</div>
                <div className="ratio-bar-track">
                  <div className="ratio-bar-fill" style={{
                    width: `${Math.min(100, (r.value / (r.ideal * 1.5)) * 100)}%`,
                    background: barColor(r.status)
                  }} />
                </div>
                <div className="ratio-bar-val" style={{ color: barColor(r.status) }}>
                  {r.value?.toFixed(2)}
                </div>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* Flags */}
      {data.flags?.length > 0 && (
        <div className="analysis-section">
          <div className="analysis-section-header">
            <span className="section-icon">🚩</span> Coaching Flags
          </div>
          <div className="analysis-section-body">
            <div className="flags-list">
              {data.flags.map((f, i) => (
                <div className={`flag-item flag-${f.type}`} key={i}>
                  <div className="flag-dot" />
                  <div className="flag-text">{f.text}</div>
                </div>
              ))}
            </div>
          </div>
        </div>
      )}

      {/* Program */}
      {data.weeklyProgram && (
        <div className="analysis-section">
          <div className="analysis-section-header">
            <span className="section-icon">📋</span> Weekly Program
          </div>
          <div className="analysis-section-body">
            <div className="analysis-prose" style={{ marginBottom: 16, color: "var(--muted)" }}>
              {data.weeklyProgram.philosophy}
            </div>
            {data.weeklyProgram.days?.map((d, i) => (
              <div className="program-day" key={i}>
                <div className="program-day-label">{d.day}</div>
                <div className="program-day-content">
                  <span style={{ color: "var(--accent)", fontWeight: 600, fontSize: "0.82rem" }}>
                    {d.focus}
                  </span>
                  <br />
                  {d.detail}
                </div>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* Next steps */}
      <div className="analysis-section">
        <div className="analysis-section-header">
          <span className="section-icon">🎯</span> Next Steps
        </div>
        <div className="analysis-section-body">
          <div className="analysis-prose">{data.nextSteps}</div>
        </div>
      </div>
    </div>
  );
}

// ─── Main App ──────────────────────────────────────────────────────────────────
export default function App() {
  const [tab, setTab] = useState("analyze");
  const [lifts, setLifts] = useState(DEFAULT_LIFTS);
  const [athlete, setAthlete] = useState({ name: "", age: "", bw: "", experience: "Intermediate" });
  const [selectedGoals, setSelectedGoals] = useState(["Strength"]);
  const [notes, setNotes] = useState("");
  const [loading, setLoading] = useState(false);
  const [analysisData, setAnalysisData] = useState(null);
  const [error, setError] = useState("");
  const [rawDebug, setRawDebug] = useState("");
  const [apiKey, setApiKey] = useState("");

  // Chat state
  const [messages, setMessages] = useState([
    {
      role: "assistant",
      content: "Hey! I'm your AI strength coach. Run an analysis first, then ask me anything — about your numbers, how to improve specific lifts, programming questions, or how to structure your training block."
    }
  ]);
  const [chatInput, setChatInput] = useState("");
  const [chatLoading, setChatLoading] = useState(false);
  const chatEndRef = useRef(null);

  useEffect(() => {
    chatEndRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages]);

  const toggleGoal = (g) => {
    setSelectedGoals(prev =>
      prev.includes(g) ? prev.filter(x => x !== g) : [...prev, g]
    );
  };

  const updateLift = (key, value) => {
    setLifts(prev => prev.map(l => l.key === key ? { ...l, value } : l));
  };

  const handleAnalyze = async () => {
    if (!apiKey.trim()) { setError("Please enter your Anthropic API key above."); return; }
    setLoading(true);
    setError("");
    setAnalysisData(null);
    setTab("analyze");
    try {
      const prompt = buildPrompt(lifts, athlete, selectedGoals, notes);
      const res = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: { "Content-Type": "application/json", "anthropic-dangerous-direct-browser-access": "true", "x-api-key": apiKey },
        body: JSON.stringify({
          model: "claude-sonnet-4-5",
          max_tokens: 4000,
          messages: [{ role: "user", content: prompt }]
        })
      });
      const rawText = await res.text();
      setRawDebug(rawText.slice(0, 500));
      if (!res.ok) throw new Error(`HTTP ${res.status}: ${rawText.slice(0,200)}`);
      const data = JSON.parse(rawText);
      if (data.error) throw new Error(data.error?.message || JSON.stringify(data.error));
      if (!data.content || !data.content.length) throw new Error("Empty content array");
      const text = data.content.map(b => b.text || "").join("");
      const start = text.indexOf("{");
      const end = text.lastIndexOf("}");
      if (start === -1 || end === -1) throw new Error(`No JSON found. Raw response: ${rawText.slice(0,300)}`);
      const parsed = JSON.parse(text.slice(start, end + 1));
      setRawDebug("");
      setAnalysisData(parsed);
    } catch (e) {
      setError(`Analysis failed: ${e.message}`);
    } finally {
      setLoading(false);
    }
  };

  const handleChat = async () => {
    const msg = chatInput.trim();
    if (!msg || chatLoading) return;
    setChatInput("");

    const userMsg = { role: "user", content: msg };
    const newMessages = [...messages, userMsg];
    setMessages(newMessages);
    setChatLoading(true);

    const systemPrompt = `You are an expert strength and conditioning coach AI assistant. Be concise, direct, and specific. 
${analysisData ? `Current athlete analysis: ${JSON.stringify(analysisData)}` : "No analysis run yet — answer general S&C questions."}
Keep responses focused and practical. Use specific numbers and percentages when relevant.`;

    try {
      const res = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: { "Content-Type": "application/json", "anthropic-dangerous-direct-browser-access": "true", "x-api-key": apiKey },
        body: JSON.stringify({
          model: "claude-sonnet-4-5",
          max_tokens: 2000,
          system: systemPrompt,
          messages: newMessages.map(m => ({ role: m.role, content: m.content }))
        })
      });
      const data = await res.json();
      const reply = data.content?.map(b => b.text || "").join("") || "No response.";
      setMessages(prev => [...prev, { role: "assistant", content: reply }]);
    } catch {
      setMessages(prev => [...prev, { role: "assistant", content: "Error reaching the API. Please try again." }]);
    } finally {
      setChatLoading(false);
    }
  };

  return (
    <>
      <style>{globalStyles}</style>
      <div className="app">
        {/* Header */}
        <header className="header">
          <div className="header-brand">
            <div className="header-title">IRONMIND AI</div>
            <div className="header-sub">S&amp;C Intelligence v3</div>
          </div>
          <div className="header-tabs">
            <button className={`tab-btn ${tab === "analyze" ? "active" : ""}`} onClick={() => setTab("analyze")}>
              Analysis
            </button>
            <button className={`tab-btn ${tab === "chat" ? "active" : ""}`} onClick={() => setTab("chat")}>
              Chat
            </button>
          </div>
        </header>

        <div className="main">
          {/* Sidebar */}
          <aside className="sidebar">
            {/* API Key */}
            <div className="sidebar-section" style={{ background: "rgba(232,255,71,0.04)", borderBottom: "1px solid rgba(232,255,71,0.15)" }}>
              <div className="section-label" style={{ color: "var(--accent)" }}>Anthropic API Key</div>
              <input
                className="form-input"
                type="password"
                placeholder="sk-ant-..."
                value={apiKey}
                onChange={e => setApiKey(e.target.value)}
                style={{ fontFamily: "DM Mono, monospace", fontSize: "0.78rem" }}
              />
              <div style={{ fontSize: "0.68rem", color: "var(--muted)", marginTop: 6, lineHeight: 1.5 }}>
                Get yours at <span style={{ color: "var(--accent)" }}>console.anthropic.com</span>. Never stored or sent anywhere except Anthropic.
              </div>
            </div>
            {/* Athlete info */}
            <div className="sidebar-section">
              <div className="section-label">Athlete Profile</div>
              <div className="form-group">
                <label className="form-label">Name</label>
                <input className="form-input" placeholder="e.g. Alex Johnson" value={athlete.name}
                  onChange={e => setAthlete(a => ({ ...a, name: e.target.value }))} />
              </div>
              <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 10 }}>
                <div className="form-group">
                  <label className="form-label">Age</label>
                  <input className="form-input" type="number" placeholder="25" value={athlete.age}
                    onChange={e => setAthlete(a => ({ ...a, age: e.target.value }))} />
                </div>
                <div className="form-group">
                  <label className="form-label">Body Wt (lbs)</label>
                  <input className="form-input" type="number" placeholder="185" value={athlete.bw}
                    onChange={e => setAthlete(a => ({ ...a, bw: e.target.value }))} />
                </div>
              </div>
              <div className="form-group">
                <label className="form-label">Experience Level</label>
                <select className="form-select" value={athlete.experience}
                  onChange={e => setAthlete(a => ({ ...a, experience: e.target.value }))}>
                  {EXPERIENCE.map(x => <option key={x}>{x}</option>)}
                </select>
              </div>
            </div>

            {/* Lifts */}
            <div className="sidebar-section">
              <div className="section-label">1-Rep Max / Performance</div>
              {lifts.map(l => (
                <div className="lift-row" key={l.key}>
                  <div className="lift-name">{l.name}</div>
                  <input
                    className="lift-input"
                    type="number"
                    placeholder="0"
                    value={l.value}
                    onChange={e => updateLift(l.key, e.target.value)}
                  />
                  <div className="lift-unit">{l.key === "pullup" ? "reps" : "lbs"}</div>
                </div>
              ))}
            </div>

            {/* Goals */}
            <div className="sidebar-section">
              <div className="section-label">Training Goals</div>
              <div className="chip-group">
                {GOALS.map(g => (
                  <button key={g} className={`chip ${selectedGoals.includes(g) ? "active" : ""}`}
                    onClick={() => toggleGoal(g)}>
                    {g}
                  </button>
                ))}
              </div>
            </div>

            {/* Notes */}
            <div className="sidebar-section">
              <div className="section-label">Additional Context</div>
              <textarea className="form-textarea"
                placeholder="Injuries, sport, competition schedule, weak points, equipment available…"
                value={notes}
                onChange={e => setNotes(e.target.value)}
              />
            </div>

            {/* Analyze button */}
            <div className="sidebar-section" style={{ borderBottom: "none" }}>
              {error && (
                <div style={{ color: "var(--danger)", fontSize: "0.78rem", marginBottom: 6, fontFamily: "DM Mono, monospace" }}>
                  ⚠ {error}
                </div>
              )}
              {rawDebug && (
                <div style={{ color: "var(--muted)", fontSize: "0.62rem", marginBottom: 10, fontFamily: "DM Mono, monospace", wordBreak: "break-all", background: "var(--surface2)", padding: "8px", borderRadius: "4px", maxHeight: "150px", overflow: "auto", lineHeight: 1.4 }}>
                  {rawDebug}
                </div>
              )}
              <button className="analyze-btn" onClick={handleAnalyze} disabled={loading}>
                {loading ? "ANALYZING…" : "RUN ANALYSIS"}
              </button>
            </div>
          </aside>

          {/* Right panel */}
          <main className="panel">
            {tab === "analyze" && (
              <>
                {loading && (
                  <div className="loading-state">
                    <div className="loading-text">Analyzing athlete data</div>
                    <div className="loading-bar-wrap"><div className="loading-bar" /></div>
                    <div style={{ fontSize: "0.75rem", color: "var(--muted)", fontFamily: "DM Mono, monospace" }}>
                      Computing lift ratios · Building program · Identifying weaknesses
                    </div>
                  </div>
                )}
                {!loading && !analysisData && (
                  <div className="empty-state">
                    <div className="empty-icon">🏋️</div>
                    <div className="empty-title">NO ANALYSIS YET</div>
                    <div className="empty-sub">
                      Enter your 1RMs on the left, set your goals, and hit Run Analysis to get a full coaching report with program recommendations.
                    </div>
                  </div>
                )}
                {!loading && analysisData && <AnalysisView data={analysisData} />}
              </>
            )}

            {tab === "chat" && (
              <div className="chat-wrap">
                <div className="chat-messages">
                  {messages.map((m, i) => (
                    <div className={`chat-msg ${m.role === "user" ? "user" : ""}`} key={i}>
                      <div className={`chat-avatar ${m.role === "user" ? "user" : "ai"}`}>
                        {m.role === "user" ? "👤" : "AI"}
                      </div>
                      <div className={`chat-bubble ${m.role === "user" ? "user" : "ai"}`}>
                        {m.content}
                      </div>
                    </div>
                  ))}
                  {chatLoading && (
                    <div className="chat-msg">
                      <div className="chat-avatar ai">AI</div>
                      <div className="chat-bubble ai">
                        <div className="typing-dots">
                          <span /><span /><span />
                        </div>
                      </div>
                    </div>
                  )}
                  <div ref={chatEndRef} />
                </div>
                <div className="chat-input-row">
                  <textarea
                    className="chat-input"
                    placeholder="Ask about your analysis, programming, lift technique, periodization…"
                    value={chatInput}
                    onChange={e => setChatInput(e.target.value)}
                    onKeyDown={e => { if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); handleChat(); } }}
                    rows={1}
                  />
                  <button className="chat-send-btn" onClick={handleChat} disabled={chatLoading || !chatInput.trim()}>
                    SEND
                  </button>
                </div>
              </div>
            )}
          </main>
        </div>
      </div>
    </>
  );
}
